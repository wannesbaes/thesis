% BS block parameters

\pgfmathsetlengthmacro{\BoxGap}{\BSBlockHeight / 100}

\pgfmathsetlengthmacro{\BSWidth}{3*\BSBlockWidth + 2*\NodeDistance + 2*\BoxGap}
\pgfmathsetlengthmacro{\BSHeight}{\BSBlockHeight + 2*\BoxGap}

% UT block parameters
\newcommand{\UTBlockWidth}{1.5cm}
\newcommand{\UTBlockHeight}{3cm}
\newcommand{\UTGapHeight}{1.60}

\pgfmathsetlengthmacro{\SingleStreamUTBlockHeight}{\UTBlockHeight / 4.5}
\pgfmathsetlengthmacro{\yShiftSingleStreamFirstBlockUT}{0.5*\UTBlockHeight - 0.5*\SingleStreamUTBlockHeight}

\pgfmathsetlengthmacro{\UTWidth}{3*\UTBlockWidth + 2*\NodeDistance + 2*\BoxGap}
\pgfmathsetlengthmacro{\UTHeight}{\UTBlockHeight + 2*\BoxGap}


% Antenna array parameters
\newcommand{\AntennaPoleHeight}{0.5cm}
\newcommand{\AntennaArmLength}{0.4cm}
\newcommand{\AntennaArmAngle}{45}

\newcommand{\AntennaLabelGapX}{2mm}
\newcommand{\AntennaLabelGapY}{1.5mm}
\newcommand{\AntennaPropagationGap}{2mm}

\pgfmathsetlengthmacro{\BSAntennaArraySep}{\BSBlockHeight/4}
\pgfmathsetlengthmacro{\UTAntennaArraySep}{\UTBlockHeight/3}

\pgfmathsetlengthmacro{\BSToAntennaGap}{\BSWidth/10}
\pgfmathsetlengthmacro{\UTToAntennaGap}{\UTWidth/10}

\pgfmathsetlengthmacro{\AntennaHeight}{\AntennaPoleHeight + \AntennaArmLength*sin(\AntennaArmAngle)}
\pgfmathsetlengthmacro{\AntennaWidth}{\AntennaArmLength*cos(\AntennaArmAngle)}

        %\draw[thick] (-0.18cm,0) -- (0.18cm,0);

    }

\newcommand{\drawUT}[2]{
    % #1 = number
    % #2 = position

    % Combiner block
    \node[ut_block]  (ut#1_combiner) at (#2) {Combiner};

    % Detector block
    \node[ut_block, right=of ut#1_combiner] (ut#1_detector) {Detector};

    % Demapper blocks
    \node[ut_block_single_stream, right=of ut#1_detector, yshift=\yShiftSingleStreamFirstBlockUT] (ut#1_demapper1) {Demapper};
    \node[ut_block_single_stream, right=of ut#1_detector, yshift=-\yShiftSingleStreamFirstBlockUT] (ut#1_demapperNt) {Demapper};
    \node (ut#1_demapperdots) at ($(ut#1_demapper1)!0.5!(ut#1_demapperNt)$) {\vdots};

    % Bit collector
    \node[ut_block, right=of ut#1_demapper1, yshift=-\yShiftSingleStreamFirstBlockUT] (ut#1_bit_collector) {Bit\\Collector};

    % UT box
    \coordinate (ut#1_northwest) at ([xshift=-\BoxGap, yshift=\BoxGap]ut#1_combiner.north west);
    \coordinate (ut#1_southwest) at ([xshift=-\BoxGap, yshift=-\BoxGap]ut#1_combiner.south west);
    \coordinate (ut#1_northeast) at ([xshift=\BoxGap, yshift=\BoxGap]ut#1_bit_collector.north east);
    \coordinate (ut#1_southeast) at ([xshift=\BoxGap, yshift=-\BoxGap]ut#1_bit_collector.south east);
    \node[draw, thick, fit=(ut#1_northwest)(ut#1_southwest)(ut#1_northeast)(ut#1_southeast), label=above right:\textbf{UT}\textsubscript{\textbf{#1}}] (ut#1_box) {};

    
    % Antenna 1
    \coordinate (ut#1_antenna1) at ($(ut#1_box.west |- ut#1_demapper1) + (-\BoxGap,0) + (-\UTToAntennaGap,0)$);
    \pic at (ut#1_antenna1) {antenna symbol};
    \node[xshift=-\AntennaLabelGapX, yshift=\AntennaLabelGapY] at (ut#1_antenna1) {$1$};
    \draw[thick] (ut#1_box.west |- ut#1_antenna1) -- (ut#1_antenna1);

    % Antenna Nr
    \coordinate (ut#1_antennaNr) at ($(ut#1_box.west |- ut#1_demapperNt) + (-\BoxGap,0) + (-\UTToAntennaGap,0)$);
    \pic at (ut#1_antennaNr) {antenna symbol};
    \node[xshift=-\AntennaLabelGapX, yshift=\AntennaLabelGapY] at ($(ut#1_antennaNr) + (-0.15,0)$) {$N_R$};
    \draw[thick] (ut#1_box.west |- ut#1_antennaNr) -- (ut#1_antennaNr);

    % Faint antennas.
    \coordinate (ut#1_antenna_dots) at ($0.5*(ut#1_antenna1) + 0.5*(ut#1_antennaNr) + 0.5*(0,\AntennaHeight)$);
    \node at (ut#1_antenna_dots) {\vdots};


    % Data flow arrows inside UT
    \draw[arrow] (ut#1_box.west |- ut#1_antenna1.east) -- (ut#1_combiner.west |- ut#1_antenna1.east);
    \draw[arrow] (ut#1_box.west |- ut#1_antennaNr.east) -- (ut#1_combiner.west |- ut#1_antennaNr.east);
    \draw[arrow] (ut#1_combiner.east |- ut#1_demapper1.west) -- (ut#1_detector.west |- ut#1_demapper1.west);
    \draw[arrow] (ut#1_combiner.east |- ut#1_demapperNt.west) -- (ut#1_detector.west |- ut#1_demapperNt.west);
    \draw[arrow] (ut#1_detector.east |- ut#1_demapper1.west) -- (ut#1_demapper1.west);
    \draw[arrow] (ut#1_detector.east |- ut#1_demapperNt.west) -- (ut#1_demapperNt.west);
    \draw[arrow] (ut#1_demapper1.east) -- (ut#1_bit_collector.west |- ut#1_demapper1.east);
    \draw[arrow] (ut#1_demapperNt.east) -- (ut#1_bit_collector.west |- ut#1_demapperNt.east);

}
