\documentclass{standalone}
\usepackage{tikz}
\usepackage[active,tightpage]{preview}
\PreviewEnvironment{tikzpicture}
\usetikzlibrary{arrows.meta, calc}

\begin{document}

% ===================
%  Layout Parameters
% ===================

% Constellation grid
\newcommand{\gridUnit}{0.9cm}         % spacing between adjacent symbols
\newcommand{\pointSize}{6pt}
\newcommand{\labelOffset}{5pt}        % vertical gap from point center to label top
\definecolor{pointColor}{RGB}{31,119,180}
\definecolor{decisionLineColor}{RGB}{214,39,40}

% Axes
\newcommand{\XaxisOverhang}{0.8cm}
\newcommand{\YaxisOverhang}{0.6cm}

% ===================
% ===================

\begin{tikzpicture}

% Component Styles
% ================

\tikzset{
    point/.style = {
        circle, inner sep=0pt,
        fill=pointColor,
        minimum size=\pointSize,
    },
    axisline/.style = {
        ->, thick
    },
    decisionline/.style = {
        dashed, decisionLineColor, thin
    },
}

% =================


% Figure Components
% =================

% Axes
\draw[axisline]
    ({-3*\gridUnit - \XaxisOverhang}, 0) --
    ({3*\gridUnit + \XaxisOverhang}, 0)
    node[right] {$\mathrm{Real}$};
\draw[axisline]
    (0, {-3*\gridUnit - \YaxisOverhang}) --
    (0, {3*\gridUnit + \YaxisOverhang})
    node[above] {$\mathrm{Imag}$};

% Axis ticks at each coordinate level
\foreach \val in {-3, -1, 1, 3} {
    \draw ({(\val)*\gridUnit},  2pt) -- ({(\val)*\gridUnit}, -2pt);
    \draw ( 2pt, {(\val)*\gridUnit}) -- (-2pt, {(\val)*\gridUnit});
}

% Constellation points and Gray-coded bit labels
% I-axis Gray code (left  to right): -3→00, -1→01, +1→11, +3→10
% Q-axis Gray code (bottom to  top): -3→00, -1→01, +1→11, +3→10
% Full label: [I-bits][Q-bits]
\foreach \ival/\ibits in {-3/00, -1/01, 1/11, 3/10} {
    \foreach \jval/\jbits in {-3/00, -1/01, 1/11, 3/10} {
        \node[point] at ({(\ival)*\gridUnit}, {(\jval)*\gridUnit}) {};
        \node[font=\scriptsize, anchor=north, yshift=-\labelOffset]
            at ({(\ival)*\gridUnit}, {(\jval)*\gridUnit}) {$\ibits\jbits$};
    }
}

% Decision region boundaries (midpoints between adjacent symbols)
\foreach \val in {-2, 0, 2} {
    \draw[decisionline] ({(\val)*\gridUnit}, {-3.75*\gridUnit}) -- ({(\val)*\gridUnit}, {3.75*\gridUnit});
    \draw[decisionline] ({-3.75*\gridUnit}, {(\val)*\gridUnit}) -- ({3.75*\gridUnit}, {(\val)*\gridUnit});
}

\end{tikzpicture}

\end{document}
