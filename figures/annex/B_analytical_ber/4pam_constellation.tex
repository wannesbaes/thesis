\documentclass{standalone}
\usepackage{tikz}
\usepackage[active,tightpage]{preview}
\PreviewEnvironment{tikzpicture}
\usetikzlibrary{arrows.meta, calc}

\begin{document}

% ===================
%  Layout Parameters
% ===================

% Constellation
\newcommand{\gridUnit}{1cm}
\newcommand{\pointSize}{6pt}
\newcommand{\labelOffset}{5pt}
\definecolor{pointColor}{RGB}{31,119,180}

% Axes
\newcommand{\XaxisOverhang}{0.9cm}
\newcommand{\YaxisHeight}{1.25cm} 

% Decision lines
\newcommand{\decisionLineHeight}{1cm}
\definecolor{decisionLineColor}{RGB}{214,39,40}

% ===================
% ===================

\begin{tikzpicture}

% Component Styles
% ================

\tikzset{
    point/.style = {
        circle, inner sep=0pt,
        fill=pointColor,
        minimum size=\pointSize,
    },
    axisline/.style = {
        ->, thick
    },
    decisionline/.style = {
        dashed, decisionLineColor, thin
    },
}

% =================


% Figure Components
% =================

% Decision region boundaries (drawn before axes so axes render on top)
% Boundaries at midpoints between adjacent symbols: -2, 0, +2 grid units
\foreach \val in {-2, 0, 2} {
    \draw[decisionline]
        ({(\val)*\gridUnit}, {-\decisionLineHeight}) --
        ({(\val)*\gridUnit},  {\decisionLineHeight});
}

% Axes
\draw[axisline]
    ({-3*\gridUnit - \XaxisOverhang}, 0) --
    ({3*\gridUnit  + \XaxisOverhang}, 0)
    node[right] {$\mathrm{Real}$};
\draw[axisline]
    (0, {-\YaxisHeight}) --
    (0,  {\YaxisHeight})
    node[above] {$\mathrm{Imag}$};

% Axis ticks at each symbol position
\foreach \val in {-3, -1, 1, 3} {
    \draw ({(\val)*\gridUnit},  2pt) -- ({(\val)*\gridUnit}, -2pt);
}

% Constellation points and Gray-coded bit labels
% Left to right: -3→00, -1→01, +1→11, +3→10
\foreach \ival/\bits in {-3/00, -1/01, 1/11, 3/10} {
    \node[point] at ({(\ival)*\gridUnit}, 0) {};
    \node[font=\small, anchor=south, yshift=\labelOffset]
        at ({(\ival)*\gridUnit}, 0) {$\bits$};
}

\end{tikzpicture}

\end{document}
