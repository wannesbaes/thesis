\documentclass{standalone}
\usepackage{tikz}
\usepackage[active,tightpage]{preview}
\PreviewEnvironment{tikzpicture}
\usetikzlibrary{arrows.meta, positioning, fit, calc, shapes.geometric}

\begin{document}

% ===================
%  Layout Parameters
% ===================

% Horizontal spacing
\newcommand{\InputToPrecoder}{0.8cm}
\newcommand{\PrecoderToChannel}{0.8cm}
\newcommand{\ChannelToNoiseAdder}{0.8cm}
\newcommand{\NoiseAdderToCombiner}{0.8cm}
\newcommand{\CombinerToOutput}{1cm}

% Vertical spacing
\newcommand{\NoiseHeight}{0.8cm}

% Multiplier size
\newcommand{\MultiplierWidth}{0.75cm}
\newcommand{\MultiplierHeight}{0.75cm}

% Adder size
\newcommand{\AdderRadius}{0.25cm}

% ===================
% ===================

\begin{tikzpicture}


% Component Styles
% ================

\tikzset{
    multiplier/.style = {
        draw, 
        rectangle,
        minimum width=\MultiplierWidth,
        minimum height=\MultiplierHeight,
        align=center
    },
    adder/.style = {
        draw, 
        circle,
        minimum size=2*\AdderRadius,
        align=center
    },
    signal/.style = {
        ->,
    }
}
% =================


% Figure Components
% =================

\node (a) at (0,0) {$\mathbf{a}$};
\node[multiplier] (precoder) at ($(a.east) + (\InputToPrecoder,0) + 0.5*(\MultiplierWidth,0)$) {$\mathbf{F}$};
\node[multiplier] (channel) at ($(precoder.east) + (\PrecoderToChannel,0) + 0.5*(\MultiplierWidth,0)$) {$\mathbf{H}$};
\node[adder] (noise_adder) at ($(channel.east) + (\ChannelToNoiseAdder,0) + (\AdderRadius,0)$) {};
\draw ($(noise_adder) + (-\AdderRadius, 0)$) -- ($(noise_adder) + (\AdderRadius, 0)$);
\draw ($(noise_adder) + (0, -\AdderRadius)$) -- ($(noise_adder) + (0, \AdderRadius)$);
\node (n1) at ($(noise_adder) + (0,\NoiseHeight)$) {$\mathbf{n}$};
\node[multiplier] (combiner) at ($(noise_adder.east) + (\NoiseAdderToCombiner,0) + 0.5*(\MultiplierWidth,0)$) {$\mathbf{G}$};
\node (y1) at ($(combiner.east) + (\CombinerToOutput,0)$) {$\mathbf{y}$};

\draw[signal] (a) -- (precoder);
\draw[signal] (precoder) -- (channel);
\draw[signal] (channel) -- (noise_adder);
\draw[signal] (n1) -- (noise_adder);
\draw[signal] (noise_adder) -- (combiner);
\draw[signal] (combiner) -- (y1);


\end{tikzpicture}

\end{document}
