\section*{System Parameters}

\begin{description}[leftmargin=3cm, labelwidth=2.7cm, labelsep=0.25em, font=\normalfont]
    \item[$K$]   Number of UTs
    \item[$N_T$] Number of transmit antennas at the BS
    \item[$N_R$] Number of receive antennas at each UT
    \item[$N_S$] Number of data streams per UT
    \item[$P_t$] Total transmit power
    \item[$N_0$] Noise power on each receive antenna
\end{description}
